# Matriz 3x3
matriz = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
]

# Suma de cada fila
for i in range(len(matriz)):
    suma_fila = sum(matriz[i])
    print("Suma fila", i, ":", suma_fila)

# Suma de cada columna
for j in range(len(matriz[0])):
    suma_columna = 0

    for i in range(len(matriz)):
        suma_columna += matriz[i][j]

    print("Suma columna", j, ":", suma_columna)
